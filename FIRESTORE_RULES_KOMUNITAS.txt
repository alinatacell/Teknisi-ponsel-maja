function isCommunityMember() {
  return isAuth() &&
    (
      isOwner() ||
      myRole() == 'karyawan' ||
      myRole() == 'employee' ||
      isAdmin()
    );
}

function isCommunityManager() {
  return isSuperAdmin() || isAdmin() || isOwner();
}

match /communityProducts/{id} {
  allow read: if isCommunityMember();
  allow create, update, delete: if isCommunityManager();
}

match /communityOrders/{id} {
  allow read: if isCommunityMember();

  allow create:
    if isCommunityMember() &&
       request.resource.data.createdBy == myUid() &&
       request.resource.data.status == 'menunggu';

  allow update: if isCommunityManager();
  allow delete: if false;
}
